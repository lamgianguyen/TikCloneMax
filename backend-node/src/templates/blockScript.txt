  // Bundle's main app Socket.IO client uses default reconnect config
  // (1s initial, 5s max) which hammers the backend after a restart.
  // Patch `window.io` once available so any `new io(...)` from the bundle
  // inherits our jittered exponential backoff. Idempotent — bundle's
  // own connect config wins if it sets the same keys explicitly. We only
  // wrap when the value is the CALLABLE io factory (typeof === 'function');
  // socketioclient.js later reassigns `window.io` to a socket INSTANCE,
  // which we leave untouched.
  (function tfPatchIODefaults(){
    var DEFAULTS = {
      reconnection: true,
      reconnectionAttempts: Infinity,
      reconnectionDelay: 2000,
      reconnectionDelayMax: 30000,
      randomizationFactor: 0.5,
      timeout: 10000,
    };
    function wrap(originalIo) {
      if (typeof originalIo !== 'function' || originalIo.__tfPatched) return originalIo;
      var wrapped = function() {
        var args = Array.prototype.slice.call(arguments);
        var lastIdx = args.length - 1;
        if (args.length === 0) {
          args.push(Object.assign({}, DEFAULTS));
        } else if (typeof args[lastIdx] === 'object' && args[lastIdx] !== null && !Array.isArray(args[lastIdx])) {
          args[lastIdx] = Object.assign({}, DEFAULTS, args[lastIdx]);
        } else {
          args.push(Object.assign({}, DEFAULTS));
        }
        // Allow both `io(...)` and `new io(...)` call styles.
        if (this instanceof wrapped) {
          return new (Function.prototype.bind.apply(originalIo, [null].concat(args)))();
        }
        return originalIo.apply(this, args);
      };
      Object.keys(originalIo).forEach(function(k) {
        try { wrapped[k] = originalIo[k]; } catch (_) {}
      });
      wrapped.prototype = originalIo.prototype;
      wrapped.__tfPatched = true;
      wrapped.__tfOriginal = originalIo;
      return wrapped;
    }
    var current = null;
    Object.defineProperty(window, 'io', {
      configurable: true,
      get: function() { return current; },
      set: function(v) {
        // Only wrap the FACTORY function (set once by socket.io.min.js).
        // Subsequent assignments (socket instances from new io(...)) pass
        // through untouched.
        current = typeof v === 'function' ? wrap(v) : v;
      }
    });
  })();

  (function tfSocketEventLogger(){
    var attached = false;
    var stats = Object.create(null);
    var poll = setInterval(function() {
      if (attached) { clearInterval(poll); return; }
      var io = window.io;
      if (!io || typeof io.on !== 'function') return;

        // Hook EVERY incoming event via onAny() — Socket.IO v4 feature.
        if (typeof io.onAny === 'function') {

            stats[event] = (stats[event] || 0) + 1;
            // Compact preview of payload
            var preview = '';
            try {
              if (payload === null || payload === undefined) preview = '';
              else if (typeof payload === 'string') preview = payload.slice(0, 80);
              else if (typeof payload === 'object') {
                var keys = Object.keys(payload).slice(0, 5);
                var pairs = keys.map(function(k) {
                  var v = payload[k];
                  if (typeof v === 'string') return k + '="' + v.slice(0, 40) + '"';
                  if (typeof v === 'number' || typeof v === 'boolean') return k + '=' + v;
                  if (Array.isArray(v)) return k + '[' + v.length + ']';
                  return k + ':obj';
                });
                preview = '{' + pairs.join(', ') + '}';
              } else { preview = String(payload).slice(0, 80); }
            } catch (e) {}
            console.log('[TF-SocketEvt #' + stats[event] + '] ' + event + ' → ' + preview);
          });
          attached = true;
          clearInterval(poll);
          console.log('[TF-SocketEvt] onAny listener attached — logging EVERY incoming event');
          window.__tfSocketStats = function() {
            console.table(stats);

    // No watchdog timers, no lastErrorAt-poll debounce, no rising-edge
    // detectors that can mis-fire during the bundle's bootstrap reload chain.
    async function poll() {
      try {
        var r = await fetch('/api/tiktok/status', { cache: 'no-store' });
        if (!r.ok) return;
        var s = await r.json();
        var connecting = Boolean(s && s.connecting);
        var connected = Boolean(s && s.connected);
