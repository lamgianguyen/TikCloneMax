/* ─────────────────────────────────────────────────────────────────────
   TTS Logs auto-trim.
   Bundle appends every TTS event to <pre id="ttsLogs"> without trimming —
   the pre grows unbounded and (per gốc HTML) has `position: absolute` +
   `max-width: 400px` so it overflows the bounding card. CSS rules above
   clip the visual overflow + force position:static, but the underlying
   text content still accumulates indefinitely. Trim text to the most
   recent N lines every 1s so DOM doesn't bloat over long live sessions.
   ───────────────────────────────────────────────────────────────────── */
(function tfTrimTtsLogs(){
  var MAX_LINES = 15;
  // Trim does actual work whenever logs append, so this timer stays
  // permanent by design (cheap DOM read + occasional textContent rewrite).
  // Documented exception to the "stabilize then clearInterval" pattern used
  // by the other long-running loops below.
  function trim(){
    try {
      var pre = document.getElementById('ttsLogs');
      if (!pre) return;
      var text = pre.textContent || '';
      var lines = text.split('\n').filter(function(l){ return l.trim(); });
      if (lines.length > MAX_LINES) {
        // Keep the most recent MAX_LINES (bundle appends to bottom).
        pre.textContent = lines.slice(-MAX_LINES).join('\n');
      }
    } catch(_){}
  }
  setInterval(trim, 1000);
})();

/* ─────────────────────────────────────────────────────────────────────
   TTS credit ledger (shared by tfForceProCredits + the /api/tts/* mocks).
   The chip = streamer's AI-TTS credit balance. User asked for REAL per-use
   decrement (instead of pinning 100k). Persist remaining in localStorage so
   both the chip force-patcher (this file, top) and tfReadCreditState (mock
   IIFE, below) read/write the SAME number. Auto-resets monthly to 100k —
   matches gốc Pro semantics ("100.000 tín dụng mỗi tháng"). Only AI/Pro voice
   generations (tts.tikfinity.com/api/tts/generate) decrement this; standard
   TikTok voices are free by design.
   ───────────────────────────────────────────────────────────────────── */
window.__tfTtsCredits = window.__tfTtsCredits || (function(){
  var KEY = 'tf_tts_credits_remaining';
  var MKEY = 'tf_tts_credits_month';
  var MAX = 100000;
  function curMonth(){ var d = new Date(); return d.getUTCFullYear() + '-' + (d.getUTCMonth() + 1); }
  function read(){
    try {
      if (localStorage.getItem(MKEY) !== curMonth()) {   // monthly refill
        localStorage.setItem(MKEY, curMonth());
        localStorage.setItem(KEY, String(MAX));
        return MAX;
      }
      var v = localStorage.getItem(KEY);
      if (v == null || v === '') return MAX;
      var n = parseInt(v, 10);
      return Number.isFinite(n) ? Math.max(0, n) : MAX;
    } catch(_) { return MAX; }
  }
  function write(n){
    try {
      localStorage.setItem(MKEY, curMonth());
      localStorage.setItem(KEY, String(Math.max(0, Math.floor(Number(n) || 0))));
    } catch(_){}
  }
  function reset(){ write(MAX); }
  return { read: read, write: write, reset: reset, MAX: MAX };
})();

/* ─────────────────────────────────────────────────────────────────────
   Overlay-settings autosave → live preview (user-requested 2026-06-02).
   The obsoverlays Customize modal writes changes via settings.set (localStorage
   ONLY — app/deobfuscated.js:66977) and its "Done" button just closes the popup
   (modules:19687) — it never POSTs. So a changed Cannon ballSize / intensity /
   color neither persisted to the backend NOR reached the preview/OBS widget
   (which read from the widgetSettings broadcast) until some unrelated
   settings.save fired. Wrap settings.set so a `widget_`-key change debounces a
   settings.save() → POST /api/updateSettings → rebuildAndBroadcast →
   widgetSettings → the preview (and OBS) widget re-applies within ~1s. The
   boot-time settings-restore burst is skipped via the `ready` flag.
   NOTE: settings + window.settings are the same singleton (both forms appear in
   the bundle), so replacing the .set METHOD intercepts every caller.
   ───────────────────────────────────────────────────────────────────── */
(function tfOverlaySettingsAutosave(){
  var ready = false;
  setTimeout(function(){ ready = true; }, 5000);   // ignore the boot settings-restore burst
  var saveTimer = null;
  var DEBOUNCE_MS = 1200;   // [save-lag fix A] widen so a whole slider drag coalesces into ONE POST
  function hasFullBag(){
    var n = 0;
    for (var i = 0; i < localStorage.length; i++) {
      if ((localStorage.key(i) || '').indexOf('setting_') === 0) { n++; if (n > 30) return true; }
    }
    return false;
  }
  function doSave(tag){
    try {
      if (!window.settings || typeof window.settings.save !== 'function') return;
      if (!hasFullBag()) return;
      // [restored fix] settings.save() returns early when !restored; force it true (belt-and-braces,
      // same as force-save) so a slider change ALWAYS persists, not just when restore() happened to run.
      window.settings.restored = true;
      window.settings.save();
    } catch(e){ /* swallow */ }
  }
  function scheduleSave(){
    if (saveTimer) { clearTimeout(saveTimer); }
    saveTimer = setTimeout(function(){ saveTimer = null; doSave('debounced'); }, DEBOUNCE_MS);
  }
  // [save-lag fix B] let force-save cancel a pending autosave so ĐƯỢC RỒI doesn't double-POST.
  window.__tfCancelAutosave = function(){ if (saveTimer) { clearTimeout(saveTimer); saveTimer = null; } };
  function install(){
    if (!window.settings || typeof window.settings.set !== 'function') return false;
    if (window.settings.__tfAutosaveWrapped) return true;
    var origSet = window.settings.set.bind(window.settings);
    window.settings.set = function(key){
      var r = origSet.apply(null, arguments);
      try {
        if (typeof key === 'string') {
          if (ready && key.indexOf('widget_') === 0) scheduleSave();
        }
      } catch(_){}
      return r;
    };
    window.settings.__tfAutosaveWrapped = true;
    return true;
  }
  var t = setInterval(function(){ if (install()) clearInterval(t); }, 500);
  setTimeout(function(){ try { clearInterval(t); } catch(_){} }, 30000);
})();

/* ─────────────────────────────────────────────────────────────────────
   tfInjectModuleTemplates (Gate 30e, 2026-05-26).

   Bundle's `tfPageloadData.modules` is an array of {name, template, hasHtml,
   hasJs, hasCss} for every page module (start, obsoverlays, goals, sounds,
   actions, ...). Bundle's native `moduleinjector.load()` would inject each
   module.template into `<div data-pageid="X">` containers.

   Our clone INTENTIONALLY keeps `appConfig.modules` empty (see blockScript
   line ~2097) so moduleinjector.load() skips its loop — that avoids the
   CDN dependency. But we ALSO need the templates injected, otherwise the
   `<div data-pageid="X">` containers stay empty and start.onVisible() has
   nothing to populate.

   This IIFE manually iterates `tfPageloadData.modules` and injects each
   `module.template` into the matching `<div data-pageid="X">` element.
   Runs before tfTriggerOverlaysOnVisible so containers exist when
   onVisible() fires.

   Gates the injection on:
   1. tfPageloadData.modules being populated
   2. The target page container existing in DOM (Vue may render it lazily)
   ───────────────────────────────────────────────────────────────────── */
(function tfInjectModuleTemplates(){
  function tryInject() {
    var pld = window.tfPageloadData;
    if (!pld || !Array.isArray(pld.modules) || pld.modules.length === 0) return false;
    var injected = 0;
    pld.modules.forEach(function(mod) {
      if (!mod || !mod.name || !mod.template || !mod.hasHtml) return;
      var container = document.querySelector('div[data-pageid="' + mod.name + '"]');
      if (!container) return;  // Vue hasn't created this page yet — skip, will retry
      // Skip if already injected (avoid double-inject + child-clobbering)
      if (container.getAttribute('data-tf-injected') === '1') return;
      try {
        container.innerHTML = mod.template;
        container.setAttribute('data-tf-injected', '1');
        injected++;
      } catch (e) {
        try { console.warn('[TF-mod-inject] failed for', mod.name, ':', e && e.message); } catch(_){}
      }
    });
    if (injected > 0) {
      try { console.log('[TF-mod-inject] injected', injected, 'module templates'); } catch(_){}
    }
    return injected;
  }
  // Poll every 300ms until all available containers are populated.
  // Vue creates `<div data-pageid="X">` lazily as user navigates, so we
  // can't do a single-shot pass at boot — need to re-check periodically.
  var iv = setInterval(tryInject, 300);
  setTimeout(function(){ clearInterval(iv); }, 60000);  // give up after 60s
  // Initial passes (Vue may already have created start container by 500ms)
  setTimeout(tryInject, 100);
  setTimeout(tryInject, 500);
  setTimeout(tryInject, 1500);
})();

/* ─────────────────────────────────────────────────────────────────────
   tfRewriteGiftImagesToCache — Route TikTok CDN gift images through
   local proxy so they hit the pre-warmed in-memory cache on the backend
   (/tiktok-img-cache/<host>/<path>) instead of going cross-origin cold.

   Context (Gate 32+33): Backend pre-warms ~200 popular gift images into
   _cdnProxyCache at boot (fire-and-forget). Electron webRequest intercept
   (Gate 33) redirects network-layer CDN fetches to the proxy. This IIFE
   is a COMPLEMENTARY JS-layer patch: it rewrites img.src in DOM so the
   browser's own src-fetch also goes same-origin, benefits from HTTP disk
   cache (Chromium), and does NOT depend on the Electron interceptor being
   active (e.g., hot-reload without Electron restart).

   Design decisions:
   - MutationObserver on document.body subtree:true / childList:true.
     Rationale: gift images appear in dropdown popup (dxSelectBox overlay,
     .dx-popup-content) which is appended to document.body outside the
     page container. Full-body scope catches gift browser + gift overlays +
     any future portal. After dropdown opens the burst is ONE mutation
     with many addedNodes — observer fires once, batch-rewrites all imgs.
     After that, few mutations → negligible overhead.
   - Idempotency: data-tf-cdn-rewritten="1" marks already-rewritten imgs.
     Prevents infinite loop (attr mutation does not trigger childList obs).
   - Initial sweep: rewrite any imgs already in DOM at install time.
   - Double-fetch: browser may have started the direct CDN fetch before
     our observer fires (img inserted, src already set). Setting img.src
     to the proxy URL cancels the in-flight CDN request and starts a new
     same-origin fetch. Acceptable tradeoff — proxy cache is warm.
     Initial sweep covers imgs inserted before observer installs.
   - Regex covers tiktokcdn.com AND tiktokcdn-us.com (US-region variant).
   - SSRF-safe: only rewrites hostnames matching *.tiktokcdn(-us)?.com.
*/
(function tfRewriteGiftImagesToCache() {
  if (window.__tfCdnRewriteInstalled) return;
  window.__tfCdnRewriteInstalled = true;

  var CDN_RE = /^(https?):\/\/([a-z0-9-]+\.tiktokcdn(?:-us)?\.com)(\/[^?#]*)?(\?[^#]*)?(#.*)?$/i;

  function rewriteImg(img) {
    if (!img || img.nodeType !== 1 || img.tagName !== 'IMG') return;
    if (img.getAttribute('data-tf-cdn-rewritten')) return;
    var src = img.getAttribute('src') || '';
    if (!src) return;
    var m = CDN_RE.exec(src);
    if (!m) return;
    var host = m[2];
    var path = (m[3] || '/').replace(/^\//, '');
    var qs   = m[4] || '';
    img.setAttribute('data-tf-cdn-rewritten', '1');
    // CRITICAL: clear bundle's onerror="opacity=0" + reset opacity BEFORE
    // swapping src. Changing img.src cancels the in-flight direct-CDN fetch,
    // which fires an 'error' event → bundle's inline onerror sets opacity:0
    // → img stays INVISIBLE even after the proxy src loads (naturalWidth>0
    // but opacity:0). Root cause of "images load but don't show".
    img.onerror = null;
    img.removeAttribute('onerror');
    if (img.style.opacity === '0') img.style.opacity = '';
    img.src = '/tiktok-img-cache/' + host + '/' + path + qs;
  }

  function sweep(root) {
    try {
      var imgs = (root || document).querySelectorAll(
        'img[src*="tiktokcdn"]:not([data-tf-cdn-rewritten])'
      );
      for (var i = 0; i < imgs.length; i++) rewriteImg(imgs[i]);
    } catch (_) {}
  }

  var obs = new MutationObserver(function (muts) {
    for (var i = 0; i < muts.length; i++) {
      var added = muts[i].addedNodes;
      for (var j = 0; j < added.length; j++) {
        var n = added[j];
        if (n.nodeType !== 1) continue;
        if (n.tagName === 'IMG') {
          rewriteImg(n);
        } else if (n.querySelectorAll) {
          var imgs = n.querySelectorAll(
            'img[src*="tiktokcdn"]:not([data-tf-cdn-rewritten])'
          );
          for (var k = 0; k < imgs.length; k++) rewriteImg(imgs[k]);
        }
      }
    }
  });

  function start() {
    if (!document.body) { setTimeout(start, 100); return; }
    sweep(document);
    obs.observe(document.body, { childList: true, subtree: true });
    try {
      console.log('[TF-cdn-rewrite] installed — gift imgs → /tiktok-img-cache/');
    } catch (_) {}
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', start, { once: true });
  } else {
    start();
  }
})();

/* ─────────────────────────────────────────────────────────────────────
   tfActivateProUI — Layer 2 trap on window.session.me.isPro (Gate 23).
   Backend (/api/me) returns isPro:true (clone is ALL-PRO by design via SQL
   local + Serial Key gate). But tfBridgeSessionMe writes session.me via
   PLAIN assignment — bundle's Vue layer / settings.restore can overwrite.
   This IIFE installs Object.defineProperty trap on:
     - window.session.me.isPro
     - window.session.me.userFeatures.isPro
   Both traps return true regardless of who writes. setInterval 2s re-installs
   on the NEW me object (tfBridgeSessionMe may swap session.me reference).
   Plus jQuery DOM activation matching bundle line 1998-2010 logic (.nopro
   hide, .appNameExtra "Pro", upgrade buttons hide).
   Pattern principle: backend is L1, but L2 client-side trap defends against
   Vue/bundle internal state divergence. See CLAUDE.md §1.00 priority L1→L5.
*/
(function tfActivateProUI(){
  function installTrap(obj, key) {
    if (!obj || typeof obj !== 'object') return 0;
    try {
      var desc = Object.getOwnPropertyDescriptor(obj, key);
      if (desc && desc.get && desc.get() === true) return 0; // already trapped
      Object.defineProperty(obj, key, {
        get: function(){ return true; },
        set: function(){ /* swallow */ },
        configurable: true,
        enumerable: true,
      });
      return 1;
    } catch (_) {
      try {
        if (obj[key] === true) return 0;
        obj[key] = true;
        return 1;
      } catch (__) { return 0; }
    }
  }
  function tick() {
    var touched = 0;
    var me = window.session && window.session.me;
    if (me) {
      touched += installTrap(me, 'isPro');
      if (!me.userFeatures) {
        me.userFeatures = { isPro: true, proInfo: null };
        touched++;
      } else {
        touched += installTrap(me.userFeatures, 'isPro');
      }
    }
    if (typeof window.$ === 'function') {
      try {
        var $nopro = window.$('.nopro:visible');
        if ($nopro.length) { $nopro.hide(); touched++; }
        var $promo = window.$('.proPromoBox:visible');
        if ($promo.length) { $promo.css('display', 'none').removeClass('shakeEffect'); touched++; }
        var $ext = window.$('.appNameExtra');
        if ($ext.length && $ext.text() !== 'Pro') {
          $ext.text('Pro').css('display', 'inline-block').addClass('proColor');
          touched++;
        }
        window.$('[class*="upgrade"], [class*="proPromo"], #upgrade-button-wrap').each(function(){
          var $el = window.$(this);
          if (/nâng cấp|upgrade/i.test($el.text()) && $el.is(':visible')) {
            $el.hide();
            touched++;
          }
        });
      } catch(_){}
    }
    return touched;
  }
  setTimeout(tick, 100);
  setTimeout(tick, 500);
  setTimeout(tick, 1500);
  setTimeout(tick, 3500);
  // Self-clearing: stop after 30 consecutive ticks (~60s) with no DOM or
  // trap changes. Tradeoff: if Vue later swaps window.session.me reference
  // after stabilization, the new object's isPro won't be re-trapped. Accepted
  // because the bridge interceptor + Pinia trap already cover most cases.
  // Function exposed on window for manual re-arm if needed.
  var _stableTicks = 0;
  var STABLE_THRESHOLD = 30;
  var _handle = setInterval(function(){
    var t = tick();
    if (t === 0) {
      _stableTicks++;
      if (_stableTicks >= STABLE_THRESHOLD) {
        try { clearInterval(_handle); } catch(_){}
        try { console.log('[TF-activate-pro-ui] stabilized after', _stableTicks, 'quiet ticks → clearInterval'); } catch(_){}
      }
    } else {
      _stableTicks = 0;
    }
  }, 2000);
  window.__tfActivateProUI = tick;
})();

/* ─────────────────────────────────────────────────────────────────────
   Live-badge accuracy patch.
   Bundle's user-profile dropdown renders a "LIVE" badge under the username
   whenever the TikTok bridge has a session (`connected: true`), regardless
   of whether the user is actually broadcasting. In gốc the badge correctly
   shows "Disconnected" when no live room is detected.
   Backend's /api/tiktok/status returns `account.isLive` + top-level
   `isLive` based on a strict check (`connected && roomId && roomInfo.data
   .status === 2`). We can't easily find the bundle's Vue render function
   for this badge (obfuscated, 'LIVE' literal appears once in string table),
   so we patch the rendered DOM directly: poll status every 5s, find the
   LIVE span in the topbar, replace text/color when isLive is false.
   ───────────────────────────────────────────────────────────────────── */
(function tfPatchLiveBadge(){
  // Approach: set `data-tf-live-state="live"` or `"disconnected"` on
  // <html> based on backend status poll. CSS rules in earlyCss.txt do the
  // visual change. This avoids fighting Vue's reactive re-render — we change
  // an attribute Vue doesn't touch, CSS handles the rest. Mutation observer
  // is removed because attribute-based CSS works without re-applying on every
  // DOM mutation (no flicker).
  var lastIsLive = null;
  function setState(isLive){
    try {
      document.documentElement.setAttribute('data-tf-live-state', isLive ? 'live' : 'disconnected');
    } catch(_){}
  }
  function refreshStatus(){
    fetch('/api/tiktok/status').then(function(r){ return r.json(); }).then(function(d){
      // Key off `connected` (the real session state), NOT the strict `isLive`
      // (connected && roomId && roomInfo.status===2). The topbar button + Pinia
      // nav corner (tf-connect.js) track `connected`, so keying this badge off
      // `isLive` re-opened the LIVE-vs-Disconnected desync whenever connected
      // but not strictly broadcasting. One canonical signal = all 3 converge.
      // (Stream-end is still caught: the bridge watchdog clears `connected` on
      // 4-min silence + streamEnd, so this won't lie "LIVE" after a real end.)
      var live = !!(d && d.connected);
      if (live !== lastIsLive) {
        lastIsLive = live;
        try { console.log('[TF-live-badge] connected=' + live + ' isLive=' + (d && d.isLive) + ' roomId=' + (d && d.roomId)); } catch(_){}
      }
      setState(live);
    }).catch(function(){});
  }
  // Default to "disconnected" until the first status poll resolves — keeps
  // the badge red on initial render instead of briefly flashing LIVE blue
  // before the poll completes.
  setState(false);
  refreshStatus();
  // Status polling cadence: 5s strikes a balance between fast UI update when
  // user starts/stops live and not hammering the backend. The bridge already
  // broadcasts `channelStatus` over Socket.IO on state change — if we wanted
  // instant update we could hook that instead; 5s poll is simpler and
  // sufficient for visual state.
  setInterval(refreshStatus, 5000);
})();

(function tfUnloadOffscreenOverlayIframes(){
  // PERF (audit 2026-06-12, biggest lag source): the SHIPPED bundle lazy-LOADS
  // overlay preview iframes (sets src from data-src on enter) but NEVER unloads
  // them off-screen — so scrolling the Overlay Library leaves all ~24 widget
  // iframes live forever (each = a socket.io connection + a 60fps render loop) →
  // "mở Lớp phủ lag màn hình". The NEWER decompiled build re-added an unload
  // (deobfuscated.js:20091-20096 src='about:blank' on leave) but it is NOT in the
  // shipped combo (about:blank count=0). Re-add it with OUR OWN IntersectionObserver:
  // blank an iframe (saving its src to data-tf-src) once it's WELL off-screen, and
  // restore on re-entry. rootMargin 600px = hysteresis so edge scroll doesn't thrash
  // load/unload. Does NOT fight the bundle's loader: on re-enter both want data-src.
  if (!('IntersectionObserver' in window)) return;
  if (window.__tfIframeUnloadInstalled) return;
  window.__tfIframeUnloadInstalled = true;
  // When an iframe is (re)loaded on scroll-in, it needs sizing. The stretch
  // override (tfStretchIframesNonBlocking) now skips unloaded iframes, so trigger
  // a debounced re-stretch after a load so the freshly-visible widget gets its
  // height. Debounced so a burst of scroll-ins collapses into one stretch pass.
  var stretchTimer = null;
  function scheduleStretch(){
    if (stretchTimer) return;
    stretchTimer = setTimeout(function(){
      stretchTimer = null;
      try { if (window.obsoverlays && window.obsoverlays.stretchIframes) window.obsoverlays.stretchIframes(); } catch(_){}
    }, 350);
  }
  var io = new IntersectionObserver(function (entries) {
    for (var i = 0; i < entries.length; i++) {
      var el = entries[i].target;
      try {
        if (entries[i].isIntersecting) {
          var restore = el.getAttribute('data-tf-src') || el.getAttribute('data-src');
          if (restore && (!el.getAttribute('src') || /about:blank/.test(el.getAttribute('src') || ''))) {
            el.removeAttribute('data-tf-src');
            el.src = restore;
            scheduleStretch();
          }
        } else {
          var cur = el.getAttribute('src');
          if (cur && !/about:blank/.test(cur)) {
            el.setAttribute('data-tf-src', cur);
            el.src = 'about:blank';
          }
        }
      } catch (_) {}
    }
  }, { root: null, rootMargin: '600px 0px', threshold: 0 });
  var seen = new WeakSet();
  function scan() {
    try {

/* ─────────────────────────────────────────────────────────────────────
   STANDALONE i18n patch installer.
   The new bundle (≥3.9MB combo/app.js) introduced 57 translation keys
   (`tts.voice_picker.*`, `tts.credits_topup.*`, `tts.daily_limit.*`, ...)
   that are baked into `downloads/vi` only. When user lands on the
   default English `downloads/index.html` the bundle's t() lookups
   return undefined → modals render the raw key (e.g. the voice-picker
   tabs show `tts.voice_picker.ai_tab` instead of "AI"). We can't easily
   patch the English HTML because the bundle reads its locale via
   `window.tfPageloadData.appConfig.localization[lang]` — that object is
   assembled inline in HTML before our injection runs.
   Strategy: poll briefly for `window.tfPageloadData` and merge the missing
   keys into every locale we know about. This keeps the modal labels
   readable regardless of which HTML variant was served. Run early
   so the merge wins the race against the bundle's first t() call.
   Patch JSON comes from `templates/i18n-patch.json`, interpolated by
   `index-html.js` via the {{i18nPatchJson}} placeholder.
   ───────────────────────────────────────────────────────────────────── */
/* ─────────────────────────────────────────────────────────────────────
   STANDALONE i18n pre-bake interceptor.
   Bundle copies `window.tfPageloadData.localization.<lang>` into its own
   vue-i18n / Composition API store AT INIT (synchronously during the first
   render pass). Any polling-based patch that runs AFTER that copy is too
   late — bundle's t() reads from the internal copy, not the live object.
   Fix: install a property interceptor on `window.tfPageloadData` BEFORE the
   inline body script that performs `window.tfPageloadData = {...}`. Our
   setter mutates the value just-assigned to add the missing keys, THEN
   stores it. By the time bundle's app.js script runs, the localization
   object already has our keys merged in — bundle's snapshot includes them.
   This works because: (1) head-injection scripts execute before body-inline
   scripts, (2) defineProperty's setter fires synchronously on assignment,
   (3) bundle reads tfPageloadData synchronously after that assignment.
   ───────────────────────────────────────────────────────────────────── */
(function tfI18nPreBake(){
  var patch = {{i18nPatchJson}};
  if (!patch || typeof patch !== 'object' || !Object.keys(patch).length) return;
  var _stored;
  function mergeInto(data){
    if (!data || !data.localization || typeof data.localization !== 'object') return;
    // Bundle hard-codes `en` as the fallback locale and crashes with
    // "Error while loading translation for en, [object Object]" when the
    // tfPageloadData.localization object lacks an `en` bucket entirely (e.g.
    // when serving vi.html which only ships `localization:{vi:{...}}`).
    // Guarantee an `en` bucket exists — use our patch JSON (EN values) as the
    // seed so bundle's en-fallback lookups don't return undefined and trigger
    // the unhandled-promise crash that leads to a black screen.
    if (!data.localization.en || typeof data.localization.en !== 'object') {
      data.localization.en = {};
      // Copy keys from any existing bucket into en as a baseline so the bundle
      // has SOMETHING to fall back to. Then patch keys add on top.
      var existingLangs = Object.keys(data.localization).filter(function(k){return k!=='en';});
      if (existingLangs.length) {
        var srcBucket = data.localization[existingLangs[0]];
        if (srcBucket && typeof srcBucket === 'object') {
          for (var srcKey in srcBucket) {
            if (Object.prototype.hasOwnProperty.call(srcBucket, srcKey)) {
              data.localization.en[srcKey] = srcBucket[srcKey];
            }
          }
        }
      }
    }
    var locs = Object.keys(data.localization);
    var inserted = 0;
    locs.forEach(function(lang){
      var bucket = data.localization[lang];
      if (!bucket || typeof bucket !== 'object') return;
      for (var k in patch) {
        if (Object.prototype.hasOwnProperty.call(patch, k) && !(k in bucket)) {
          bucket[k] = patch[k];
          inserted++;
        }
      }
    });
    try { console.log('[TF-i18n-prebake] intercepted assign — merged', inserted, 'into locales', locs); } catch(_){}
  }
  try {
    // If tfPageloadData already exists (e.g. on bundle hot-reload where the
    // setter trap may have been wiped), patch the live object and skip the
    // defineProperty install (can't redefine without configurable:true on
    // existing prop — and avoiding a throw matters more than installing the
    // trap a second time).
    if (Object.getOwnPropertyDescriptor(window, 'tfPageloadData')) {
      mergeInto(window.tfPageloadData);
      return;
    }
    Object.defineProperty(window, 'tfPageloadData', {
      configurable: true,
      enumerable: true,
      get: function(){ return _stored; },
      set: function(v){ mergeInto(v); _stored = v; }
    });
  } catch(e){
    try { console.warn('[TF-i18n-prebake] install failed:', e && e.message); } catch(_){}
  }
})();

(function tfI18nPatch(){
  var patch = {{i18nPatchJson}};
  if (!patch || typeof patch !== 'object' || !Object.keys(patch).length) return;
  function apply(){
    try {
      var pld = window.tfPageloadData;
      // Bundle ≥3.9MB moved localization OUT of appConfig: it now lives at
      // `tfPageloadData.localization.<lang>` (top-level), not inside appConfig.
      // The old nested path is gone — patching it does nothing.
      var loc = pld && pld.localization;
      if (!loc) return false;
      // Only patch locales that already have data; don't create empty buckets
      // (bundle's t() looks up by current locale only, so creating empty `vi`
      // on the English page would hide keys behind that empty bucket if the
      // bundle were to switch locale at runtime).
      var langs = Object.keys(loc);
      var merged = 0;
      langs.forEach(function(lang){
        var bucket = loc[lang];
        if (!bucket || typeof bucket !== 'object') return;
        for (var k in patch) {
          if (Object.prototype.hasOwnProperty.call(patch, k) && !(k in bucket)) {
            bucket[k] = patch[k];
            merged++;
          }
        }
      });
      try { console.log('[TF-i18n-patch] merged', merged, 'key-insertions into locales', langs); } catch(_){}
      return true;
    } catch(e){
      try { console.warn('[TF-i18n-patch] apply failed:', e && e.message); } catch(_){}
      return false;
    }
  }
  if (apply()) return;
  // tfPageloadData is set inline in the document b-o-d-y AFTER our head
  // injection runs. Note: never write the literal lower-case b o d y opening
  // tag here because the surrounding HTML build does a /body-tag/i replace and
  // would otherwise inject the entire b o d y template INSIDE this comment,
  // breaking the script tag.
  // Poll up to 5s with a tight 30ms interval so the merge lands before any
  // Vue component reads its t() bindings on first render.
  var t0 = Date.now();
  var iv = setInterval(function(){
    if (apply() || Date.now() - t0 > 5000) {
      try { clearInterval(iv); } catch(_){}
    }
  }, 30);
})();

/* ─────────────────────────────────────────────────────────────────────
   tfPatchSwitchLanguage — wrap localization.switchLanguage to also set
   the backend's tf_locale cookie before window.location.reload().

   Bundle's language picker (Vue) calls `await localization.switchLanguage(lang)`
   then `window.location.reload()`. The original switchLanguage only:
     - updates localization.languageCode + settings.set('language', lang)
     - fetches /config/localization/<lang>.json (or uses inline data)
   It does NOT set any cookie/header that survives reload AT THE BACKEND.
   On reload, our backend's detectLang() reads tf_locale cookie → maps to
   served HTML variant. Without the cookie, backend keeps serving the same
   `downloads/<lang>` file, so tfPageloadData.localization only has the OLD
   locale's bucket. Bundle init + tfI18nPreBake fall back to copying that
   bucket into the en slot, so vue-i18n's English messages contain VN text
   → UI stays Vietnamese even though languageCode='en'.

   Fix: intercept switchLanguage to set tf_locale + tf_lang cookies. After
   reload, backend serves the correct localized HTML and the bundle picks up
   the right localization bucket on init.
   ───────────────────────────────────────────────────────────────────── */
(function tfPatchSwitchLanguage(){
  // Must cover ALL 12 langs bundle's picker offers — backend's TF_LOCALE_TO_LANG
  // accepts the inverse mapping. If a lang code isn't in this table, default to
  // 'EN' would make backend serve the English HTML (wrong locale) even though
  // bundle's internal languageCode is correct.
  var LANG_TO_LOCALE = {
    vi: 'VN', de: 'DE', es: 'ES', en: 'EN',
    id: 'ID', ja: 'JA', ko: 'KO', ms: 'MS',
    th: 'TH', tl: 'TL', tr: 'TR', 'pt-br': 'BR'
  };
  function tryInstall() {
    var loc = window.localization;
    if (!loc || typeof loc.switchLanguage !== 'function') return false;
    if (loc.__tfSwitchPatched) return true;
    var origSwitch = loc.switchLanguage;
    loc.switchLanguage = function tfSwitchLanguageWrapped(langCode) {
      try {
        var code = String(langCode || '').toLowerCase();
        // Allowlist check: only write cookies if code is a known LANG_TO_LOCALE
        // key. Reject + log otherwise to prevent cookie-injection via crafted
        // langCode containing ';' or attribute breakouts.
        if (!Object.prototype.hasOwnProperty.call(LANG_TO_LOCALE, code)) {
          try { console.warn('[TF-i18n-switch] rejected unknown langCode:', langCode); } catch(_){}
          return origSwitch.call(loc, langCode);
        }
        var locale = LANG_TO_LOCALE[code];
        var maxAge = 31536000;
        // tf_lang uses the validated lowercase allowlist key. Backend's regex
        // accepts hyphenated codes like 'pt-br'.
        document.cookie = 'tf_locale=' + locale + '; Path=/; Max-Age=' + maxAge + '; SameSite=Lax';
        document.cookie = 'tf_lang=' + code + '; Path=/; Max-Age=' + maxAge + '; SameSite=Lax';
        try { console.log('[TF-i18n-switch] set tf_locale=' + locale + ' tf_lang=' + code); } catch(_){}
      } catch (_) {}
      return origSwitch.call(loc, langCode);
    };
    loc.__tfSwitchPatched = true;
    try { console.log('[TF-i18n-switch] localization.switchLanguage patched'); } catch(_){}
    return true;
  }
  if (tryInstall()) return;
  var t0 = Date.now();
  var iv = setInterval(function(){
    if (tryInstall() || Date.now() - t0 > 30000) {
      try { clearInterval(iv); } catch(_){}
    }
  }, 100);
})();

      // broadcast. Prevents a reload storm when several template files are
      // saved in quick succession (each save → debounced broadcast → reload).
      // Without this, the kill-switch in reloadGuard cannot stop the loop
      // because dev auto-reload historically bypassed Location.prototype
      // and went through __tfOrigReload directly.
      try {
        // Throttle is in-memory (per page load) — does NOT persist across
        // reloads. After a reload completes, the in-memory `__tfReloadedAt`
        // gets re-initialised, so the next legitimate broadcast (e.g. user
        // saves another template right after) is honoured immediately.
        // This avoids the "save edits → no auto-reload → must Ctrl+R" trap.
        var lastTs = window.__tfReloadedAt || 0;
        if (lastTs && (Date.now() - lastTs) < 2000) {
          console.warn('[TF-DEV] reload broadcast ignored (last was ' + (Date.now() - lastTs) + 'ms ago)');
          triggered = false;
          return;
        }
        window.__tfReloadedAt = Date.now();
      } catch (_) {}
      console.log('[TF-DEV] template changed — auto-reload');
      // Go through the regular location.reload() so reloadGuard's
      // sliding-window kill-switch can still freeze the page if a real
      // reload loop somehow develops. Bypassing reloadGuard via
      // __tfOrigReload (previous behaviour) made dev auto-reload
      // unstoppable.
      try { location.reload(); }
      catch (e) { if (typeof window.__tfOrigReload === 'function') window.__tfOrigReload(); }
    }

    function attach(sock, label) {
      if (!sock || typeof sock.on !== 'function' || sock.__tfDevReloadAttached) return;
      sock.__tfDevReloadAttached = true;
      try {
        sock.on('tf-dev-reload', doReload);
        console.log('[TF-DEV] auto-reload listener attached to', label);
      } catch (e) {}
    }

    function rescan() {
      try {
        Object.keys(window).forEach(function(k){
          try {
            var v = window[k];
            if (v && typeof v === 'object' && typeof v.on === 'function'
                && typeof v.emit === 'function' && 'connected' in v && v.nsp) {
              attach(v, 'window.' + k);
            }
          } catch(_){}
        });
      } catch(_){}
    }
    rescan();
    setTimeout(rescan, 500);
    setTimeout(rescan, 1500);
    setTimeout(rescan, 4000);
    setTimeout(rescan, 10000);

    // Wrap io() factory so any future socket the bundle creates also gets
    // the dev-reload listener. Idempotent via __tfDevReloadFactoryHooked.
    if (typeof window.io === 'function' && !window.io.__tfDevReloadFactoryHooked) {
      var origIo = window.io;
      var wrapped = function() {
        var sock = origIo.apply(this, arguments);
        try { attach(sock, 'io() factory'); } catch(_){}
        return sock;
      };
      Object.keys(origIo).forEach(function(k){ try { wrapped[k] = origIo[k]; } catch(_){} });
      wrapped.__tfDevReloadFactoryHooked = true;
      try { window.io = wrapped; } catch(_){}
    }

    // No watchdog timers, no lastErrorAt-poll debounce, no rising-edge
    // detectors that can mis-fire during the bundle's bootstrap reload chain.
    async function poll() {
      try {
        var r = await fetch('/api/tiktok/status', { cache: 'no-store' });
        if (!r.ok) return;
        var s = await r.json();
        var connecting = Boolean(s && s.connecting);
        var connected = Boolean(s && s.connected);
        var username = (s && (s.failedUsername || s.username)) || '';

        else if (connecting) setStatusPill('connecting', username);

        // (streamEnd, watchdog kill, or auto-reconnect giving up) used to flip
        // the pill silently. Toast on the falling edge instead. Armed only AFTER
        // we've seen a live session (so the page-load 'disconnected' baseline
        // never fires) and skipped while `connecting` (auto-reconnect in flight)
        // so transient self-healing blips stay quiet — only a drop that does NOT
        // recover notifies. A success toast pairs the disconnect one on recovery.
        var P = window.__tfPopup;
        if (connected) {
          if (P.connLostShown) { tfConnToast('✅ Đã kết nối lại TikTok LIVE', 'success'); P.connLostShown = false; }
          P.everLive = true;
        } else if (!connecting && P.everLive && !P.connLostShown) {
          P.connLostShown = true;
          tfConnToast('⚠️ Đã ngắt kết nối TikTok LIVE — bấm "Kết nối" để thử lại.', 'warning');

                  position: { my: 'top center', at: 'top center', offset: '0 78' } },
                type, 5000);
              return;
            }
          } catch (e) {}
          // Fallback toast if DevExpress.ui.notify isn't ready yet.
          try {
            var t = document.createElement('div');
            t.textContent = message;
            t.style.cssText = 'position:fixed;top:78px;left:50%;transform:translateX(-50%);z-index:2147483646;'

            (document.body || document.documentElement).appendChild(t);
            setTimeout(function () { try { t.remove(); } catch (e) {} }, 5000);
          } catch (e) {}

        // it can be MISSED — dropped frame, socket not yet connected, or a
        // connect path that didn't emit). `lastErrorAt` only advances on a
        // USER-initiated attempt (userClick), so this stays silent for auto /
        // bootstrap reconnects. Baseline on first poll → never fire on a stale
        // pre-load error. Debounced against the socket path so we never double.
        var errAt = Number(s && s.lastErrorAt) || 0;
        if (baselineErrorAt < 0) {
          baselineErrorAt = errAt;
        } else if (errAt > baselineErrorAt && !connecting && !connected) {
          baselineErrorAt = errAt;
          if (Date.now() - pageLoadTime > STARTUP_GRACE_MS &&
              Date.now() - lastPopupShownTime > POPUP_DEBOUNCE_MS) {
            lastPopupShownTime = Date.now();
            showPopup((s && s.lastError) || 'Unable to connect to TikTok LIVE.', username);
          }
        }
      } catch (e) {}
    }

    // Single popup trigger: socket `connectFailed` event from backend.
    // Backend only emits this when userClick: true (i.e. the user pressed

    function normalizeMountTarget(target) {
      if (!target) return target;
      if (target.nodeType === 1) return target;
      var raw = String(target);
      var trimmed = raw.trim();
      if (!trimmed) return trimmed;
      var byId = document.getElementById(trimmed.replace(/^#/, ''));

    function clearPendingMount() {
      if (pendingMountTimer) {
        clearInterval(pendingMountTimer);
        pendingMountTimer = null;
      }
    }

    function ensureApp() {
      if (app) return app;
      if (typeof VueNavigation === 'undefined' || !VueNavigation['default'] || typeof Vue === 'undefined' || !Vue.createApp) {
        return null;
      }

      console.log('[TF-DEBUG] createNavigation: using real VueNavigation component');
      app = Vue.createApp(VueNavigation['default'], props || {});
      if (typeof VueNavigation['setup'] === 'function') {
        try { VueNavigation['setup'](app); } catch (e) { console.warn('[TF-DEBUG] VueNavigation.setup error:', e); }
      }
      return app;
    }

    function mountRealApp(target) {
      var realApp = ensureApp();
      if (!realApp) return false;

      ensureLayoutVisible();

      if (realApp.__tfMounted) {
        return true;
      }

      var normalizedTarget = normalizeMountTarget(target);
      if (!normalizedTarget) {
        return false;
      }

      try {
        realApp.mount(normalizedTarget);
        realApp.__tfMounted = true;
        realApp.__tfMountTarget = normalizedTarget;
        clearPendingMount();
        ensureLayoutVisible();
        return true;
      } catch (e) {
        console.warn('[TF-DEBUG] createNavigation.mount failed:', e);
        return false;
      }
    }

    proxy = {
      mount: function(target) {
        pendingMountTarget = target;
        ensureLayoutVisible();

        if (mountRealApp(target)) {
          return app;
        }

        clearPendingMount();
        mountAttemptCount = 0;
        pendingMountTimer = setInterval(function() {
          mountAttemptCount += 1;
          if (mountRealApp(pendingMountTarget)) {
            return;
          }
          if (mountAttemptCount >= 120) {
            clearPendingMount();
            console.warn('[TF-DEBUG] createNavigation: timed out waiting for VueNavigation');
          }
        }, 50);

        return proxy;
      },
      unmount: function() {
        clearPendingMount();
        if (app && app.__tfMounted && typeof app.unmount === 'function') {
          try { app.unmount(); } catch (e) { console.warn('[TF-DEBUG] createNavigation.unmount failed:', e); }
          app.__tfMounted = false;
        }
      }
    };

    return proxy;
  };

  // Bundle ≥3.9MB filters voices via `voice.id`, `voice.name`, `voice.provider`

  function tfNormalizeMockVoice(v) {
    var vendorId = v.vendorId || '';
    var voiceId = v.voiceId || '';
    var provider = vendorId === 'ttsm' ? 'ai' : (vendorId === 'polly' ? 'polly' : vendorId);
    // Bundle (window.aiTts.resolveVoiceConfigFromId) requires AI voice IDs
    // to be prefixed with `tts_api__` (the value of window.aiTts.voiceIdPrefix
    // at runtime). Without the prefix the resolver returns null → bundle's
    // loader skips the voice → AI tab stays empty. Polly (free) voices use a
    // different format and are not parsed through this resolver.
    var idBase = vendorId && voiceId ? vendorId + '__' + voiceId : (voiceId || vendorId);
    var id = (provider === 'ai' && idBase) ? ('tts_api__' + idBase) : idBase;
    return Object.assign({}, v, {
      id: id,
      name: v.displayName || v.voiceName || '',
      provider: provider,
    });

  // Real per-use TTS credit decrement (user-requested 2026-06-02). Delegates
  // to the shared ledger (window.__tfTtsCredits, defined near tfForceProCredits
  // at the top of this file) so the chip force-patcher and these /api/tts/*
  // mocks read/write ONE number. Auto-resets to 100k monthly (gốc Pro

  function tfReadCreditState(){
    return (window.__tfTtsCredits && window.__tfTtsCredits.read)
      ? window.__tfTtsCredits.read() : 100000;
  }
  function tfWriteCreditState(n){
    if (window.__tfTtsCredits && window.__tfTtsCredits.write) window.__tfTtsCredits.write(n);
  }
  function tfBuildQuotaPayload(remaining){
    var nowDate = new Date();
    // Subscription credits reset MONTHLY per gốc Pro semantics ("100.000
    // tín dụng mỗi tháng"). Next reset = 1st of next month.
    var nextMonth = new Date(Date.UTC(
      nowDate.getUTCFullYear(),
      nowDate.getUTCMonth() + 1,
      1, 0, 0, 0
    ));
    var nextSec = Math.floor((nextMonth.getTime() - nowDate.getTime()) / 1000);
    return {
      exceeded: false, // Pro never exceeded — always has 100k
      // CRITICAL: must be 'sub_credits' (NOT 'subscription'). Bundle's
      // applyAiCreditsFromApiUser (decompiled/modules/deobfuscated.js:4621)
      // only sets tts.proCredits when mode === 'sub_credits'. 'subscription'
      // falls through → tts.proCredits stays 0 → chip shows 25 freeMessages.
      currentUsageMode: 'sub_credits',
      currentUsageCurrency: 'characters',
      subscriptionCreditsRemaining: remaining,
      subscriptionCreditsTotal: 100000,
      purchasedCreditsRemaining: 0,
      purchasedCreditsTotal: 0,
      // Keep free quota non-zero so bundle does not auto-open the
      // "daily limit exceeded" modal during Pro AI voice initialization.
      freeRequestsRemaining: 25,
      freeRequestsTotal: 25,
      nextResetAt: nextMonth.toISOString(),
      nextResetSeconds: nextSec,
    };
  }
  function tfHandleTtsGenerate(url, opts) {
    if (!/tts\.tikfinity\.com\/api\/tts\/generate(\/|\?|$)/.test(url)) return null;
    var body = {};
    try {
      if (opts && opts.body) body = typeof opts.body === 'string' ? JSON.parse(opts.body) : opts.body;
    } catch(_) {}
    var charCount = String(body.text || '').length;
    var remaining = tfReadCreditState();
    var cost = Math.max(1, charCount); // 1 credit per character (gốc semantics)
    var newRemaining = Math.max(0, remaining - cost);
    tfWriteCreditState(newRemaining);
    try { console.log('[TF-tts-gen]', 'voice=' + (body.voiceId || body.voice || '?'),
      'chars=' + charCount, 'cost=' + cost,
      'credits=' + remaining + '→' + newRemaining); } catch(_){}
    // Placeholder audio URL — user maps AI voiceId → real TikTok TTS voice
    // later. Returning a non-functional URL lets bundle complete its flow
    // (update quota state, refresh chip count) without throwing.
    var payload = {
      statusCode: 201,
      message: 'Success',
      data: {
        result: {
          fromCache: false,
          audioUrl: '/api/tts/generate?voice=en_us_002&text=' + encodeURIComponent(String(body.text||'').slice(0,200)),

  // The clone routes TikTok connect via tf-connect.js (REST -> bridge), which
  // BYPASSES the bundle's native connector. So broadcastlistener.onChat — the
  // entry point that drives the in-app Chat page + TTS — never gets fed, and TTS

  // CHAT-ONLY on purpose: onChat re-distributes via distributeEvent("chat") which

  // emitWsEvent, so there is no double-fire of widgets/DAPI. Gift/like are NOT

  (function tfChatFeedToModules(){
    function connectFeed(){
      if(window.__tfChatFeedStarted) return;
      if(typeof window.io !== 'function') return;
      if(!window.broadcastlistener || typeof window.broadcastlistener.onChat !== 'function') return;
      window.__tfChatFeedStarted = true;
      var s;
      try {
        s = window.io({ transports:['websocket'], upgrade:false, reconnection:true,
          reconnectionDelay:2000, reconnectionDelayMax:30000, query:{ appType:'controlpage', shared:false } });
      } catch(e){ window.__tfChatFeedStarted = false; return; }
      function tlog(tag, msg){
        try {
          fetch('/api/_dev/clientlog', { method:'POST', headers:{ 'Content-Type':'application/json' },
            body: JSON.stringify({ tag: tag, msg: String(msg).slice(0, 300) }) });
        } catch(e){}
      }
      s.on('connect', function(){
        try {
          var cid = (window.session && window.session.channelId) || 1;
          s.emit('setContext', { channelId: cid, appType:'controlpage' });
          tlog('FEED', 'chat feed connected cid=' + cid + ' tiktok=' + ((window.session && window.session.tiktokUsername) || localStorage.getItem('setting_tiktokname') || '?'));
        } catch(e){}
      });
      s.on('chat', function(d){
        try {
          if(d && typeof d.comment === 'string'){
            tlog('TTS-CHAT', (d.nickname || d.uniqueId || '?') + ': ' + d.comment);
            if(window.broadcastlistener && typeof window.broadcastlistener.onChat === 'function'){
              window.broadcastlistener.onChat(d);
            }
          }
        } catch(e){}
      });
      // LOG-ONLY (do NOT feed onGift — it re-emits to DAPI = double overlay).
      s.on('gift', function(d){
        try {
          if(d) tlog('GIFT', (d.nickname || d.uniqueId || '?') + ' sent ' + (d.giftName || d.describe || 'gift') + ' x' + (d.repeatCount || d.repeat_count || 1));
        } catch(e){}
      });
    }
    function loop(){
      if(window.__tfChatFeedStarted) return;
      if(typeof window.io === 'function' && window.broadcastlistener
         && typeof window.broadcastlistener.onChat === 'function'){ connectFeed(); return; }
      if(typeof window.io !== 'function' && !window.__tfChatFeedIoLoading){
        window.__tfChatFeedIoLoading = true;
        try {
          var sc = document.createElement('script');
          sc.src = '/js/lib/socket.io.min.js';
          (document.head || document.documentElement).appendChild(sc);
        } catch(e){}
      }
      setTimeout(loop, 800);
    }
    loop();
  })();
